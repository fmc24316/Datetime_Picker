export default { async fetch(request, env, ctx) { // 記錄收到的請求方法與網址 console.log(`[Request] 收到前端請求，URL: ${request.url}, Method: ${request.method}`); // 設定 CORS 標頭，允許你的 GitHub Pages 前端跨域呼叫此 API const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET,HEAD,POST,OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }; // 攔截並處理瀏覽器發出的 OPTIONS 預檢請求 if (request.method === "OPTIONS") { console.log("[OPTIONS] 處理 CORS 預檢請求"); return new Response(null, { headers: corsHeaders }); } try { // 目標租車系統 API 的實際網址 const targetUrl = 'https://www.yamicar.com.tw/MbrJoin/number_FmcGrp_Api.php'; console.log(`[Fetch] 準備呼叫目標 API: ${targetUrl}`); // 使用 POST 發送請求，以符合 fetch API 規範並順利傳遞 JSON Body 參數 const response = await fetch(targetUrl, { method: 'POST', headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' }, body: JSON.stringify({ action: "getNoWork" }) }); console.log(`[Fetch] 目標 API 回應狀態碼: ${response.status}`); // 檢查 HTTP 狀態碼是否正常，若異常則回傳空陣列讓前端繼續執行 if (!response.ok) { console.log(`[Error] 目標 API 狀態碼異常`); return new Response(JSON.stringify([]), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } // 取得原始回傳文字內容 const textData = await response.text(); console.log(`[Fetch] 目標 API 回傳原始內容前 200 字元: ${textData.substring(0, 200)}`); let formattedDates = []; // 嘗試解析 JSON 並清洗資料結構 try { const apiData = JSON.parse(textData); console.log(`[Data] 成功解析 JSON，確認回傳標記 FLG: ${apiData.FLG}`); // 確保資料結構符合預期後，提取不可營業日字串轉換為一維陣列 if (apiData.FLG === "T" && apiData.RES_DATA && Array.isArray(apiData.RES_DATA.DATE_S)) { formattedDates = apiData.RES_DATA.DATE_S.map(item => item.NO_DATE); console.log(`[Data] 轉換後的純日期陣列:`, JSON.stringify(formattedDates)); } else { console.log(`[Warning] 資料結構不如預期，回傳空陣列`); } } catch (parseError) { console.log(`[Error] 無法將回傳內容解析為 JSON: ${parseError.message}`); } // 將清洗後的乾淨陣列回傳給前端 return new Response(JSON.stringify(formattedDates), { headers: { ...corsHeaders, "Content-Type": "application/json" } }); } catch (error) { // 捕捉任何未預期的系統例外錯誤 console.log(`[Exception] 發生例外錯誤: ${error.message}`); return new Response(JSON.stringify([]), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } } };
